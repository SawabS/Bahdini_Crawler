           كه‌كو، وه‌كو ئه‌م دبینن دنیا گه‌له‌ك پێشدا چوویه‌ و جیهانگه‌ریێ هن پرێن ژ هه‌ژى دناڤبه‌را ملله‌ت و وه‌لاتان دا چێكرنه‌، ل هن جهان ژى ته‌ڤ سنۆر خاڤ كرنه‌ یان ژى راكرنه‌. به‌لێ مخابن ل ده‌ڤه‌را مه‌ كو رۆژهه‌لاتا ناڤینه‌ ئه‌و سنۆر هه‌ر ب ترس مانه‌ و ب ساناهى ناهێن ده‌ربازكرن، هن ژێ ئه‌گه‌ر راستى ژى نه‌ به‌، لێ دهشێ مه‌دا نه‌لڤ مانه‌، باسكرن ونێزیكبوونا وان گه‌له‌ك چه‌تنه‌. لێ كه‌سێ ره‌وشه‌نبیر هه‌ر تم پێشه‌نگه‌ ژ بۆ شكاندنا سنۆران و ب جان و جه‌سه‌د ژ سنۆران ده‌رباز دبن. پرسا من دڤێ بكم: وه‌ سنۆره‌ك بڕى یه‌ و ددلێ دهۆكا ئازادى یێ دا مێهڤانن، هوون هه‌ست ب چ دكه‌ن؟ هوون ژبوونا ڤسته‌ڤالێ دخوازن چ بێژن؟
ب: ل به‌رى ئه‌م سۆپاسدارى وه‌نه‌ ژ بۆ كو وه‌ ئه‌م ڤه‌خواندنه‌ و ل ڤر كرنه‌ میڤان ل مالا خوه‌ كو مالا مه‌یه‌. راسته‌ ئه‌م یێ ل باشوور، لێ ئه‌ز ب خوه‌ وه‌ك شاعره‌ك ڤى وه‌لاتى دابه‌ش ناكم، نابێژم ئه‌ڤ باكووره‌ وئه‌ڤ باشووره‌. به‌لێ پا راستیه‌ك ژیانێ یه‌ كو كوردستان پارچه‌ بوویه‌. جنابێ وه‌ژى ئانى زمان كو سنۆر ب ساناهى وهێسانى نكارن ده‌رباز ببن، نه‌مازه‌ دناڤبه‌را هن لایان دا. لێ وه‌كو هوون ژى دپه‌ژرینن ژ بۆ هونه‌رمه‌ندان، ژ بۆ نڤیسكاران سنۆر نینن. د ئاخفتنا خوه‌ یا كورت ده‌ ژى من دیار كر؛ چ هێز ...چ ركه‌ به‌رێ نكارىیه‌ ده‌نگێ فه‌قییێ ته‌یران كو نه‌گهیژه‌ ڤر، یان ژى شعرێن جه‌گه‌رخوین و مه‌لایێ جزیرى نه‌گهێن ڤر. یان ژى یێن خانى و كۆیى نه‌گهه‌ ڤر.

           ئه‌ڤرۆ وه‌ك ئه‌م دبینن پڕێن ره‌وشه‌نبیرى گه‌له‌ك د گرنگن ژ بۆنا د هه‌ڤگه‌هشتنێ و هه‌ڤ قه‌بوولكرنێ. هه‌موو ملله‌تێن دنیایێ بگشتى و ملله‌تێ مه‌ ب تایبه‌تى گه‌له‌ك پێدڤى بڤان پرایه‌. چاخ هاتیه‌ هه‌ڤ قه‌بوول بكن. مینا: ته‌ قه‌بوول بكم وه‌ك تو هه‌یى، تو ژى من قه‌بوول بكى وه‌كى كو ئه‌ز هه‌مه‌. دڤێ ئه‌ڤ پر بێن چێكرن. یانى هوون ڤان ڤسته‌ڤال و جڤینێن ب ڤى ئاوایى دبینن كو تێرا هندێ بكن ئه‌ڤ پر پێ بێنه‌ چێكرن؟ كو ئه‌م كورد ژ نێزڤه‌ هه‌ڤ ناس بكن؟ نه‌ هما ته‌نێ ستران و هه‌لبه‌ستا مه‌ بگه‌هه‌ هه‌ڤ، ئه‌ڤه‌  تا چه‌ندێ دكاره‌ ئارمانجا  مه‌ ب هنگێڤه‌؟
ب: به‌لێ ئه‌ز دگهێژم، ئه‌ڤ ده‌رفه‌تن. ده‌رفه‌ت ل ده‌ستپێكا وان تێكلیێن كو دمێژ یێ مرۆڤى ده‌ گه‌ورنه‌. ئه‌ز باوه‌رم وێ تێرێ نه‌كن، ئه‌ڤ ڤسته‌ڤال و شاهى و تشتێ ب ڤى ره‌نگى. لێ ڤسته‌ڤالا دهۆك یا ره‌وشه‌نبیرى یا یه‌كه‌م ده‌رنه‌كه‌تبا ئه‌ڤ ژى نه‌د بوو، گه‌ره‌ك ئه‌ڤ ببه‌ كه‌له‌پۆره‌ك دپێشه‌رۆژێ ده‌ ژى، چونكى ئه‌ڤ نه‌با پسیاره‌ك هۆ وێ نه‌هاتبا كرن. ژ پسیاران ب ورده‌تر كو هه‌ر چار په‌رچان سنۆران برۆخینێ، دگه‌ل ڤان پسیارا په‌یوه‌ندیێن كو یه‌ك ب یه‌ك هۆنه‌رمه‌ند، ئان ژى سازى ژوان سازیان یا یه‌كى نڤیسكار به‌ بگهن هه‌ڤ. ئه‌ڤ نڤیسكاره‌ كو ببت ب رێكا شاهى و رۆژێن ب ڤى ره‌نگێ پسیارێ، دڤێ رۆژه‌كى به‌رى رۆژه‌كێ كورد ژ بوونا كو سنۆرین هه‌یێن خوه‌زایى، ب هه‌لبه‌ستا جزیرى، ب هه‌لبه‌ستا خانى، یان

جه‌گه‌رخوین. یێن كو رۆخیانه‌. ب تێكلیه‌كا راسته‌قینه‌تر ڤان سنۆران هه‌لوه‌شینن، گیانێ خوه‌ ب هه‌ڤڕه‌ پاكژ بكن. چىیه‌؟ ئه‌و ژ بۆ كو ل سه‌ر ئاخا ئازاد كورد، ب تایبه‌ت نڤیسكاره‌ك یان ره‌وشه‌نبیره‌ك بكار بێ په‌یڤا ئه‌ده‌بیاتا نه‌ته‌وى یا كوردى بینى زمێن. ئه‌گه‌ر مه‌به‌ستا وى ژ ئه‌ده‌بیاتا قه‌تێ وى بێ یان ژ چه‌ند زاراڤه‌یان بێ بلا ببێ، مه‌به‌ست ئه‌و به‌ مرۆڤ نكاره‌ بێژه‌ ئه‌و ئه‌ده‌بیات ئه‌ده‌بیاتا نه‌ته‌وى یه‌.

           ته‌قالا زاراڤه‌یان كر. گه‌لۆ ل سه‌ر ڤان ئاسته‌نگێن هه‌یین ژ سنۆران، ژ پارچه‌بوون و داگیركرنا كوردستانێ كو ئاسته‌نگ ژ گه‌لێ مه‌ڕا ده‌ردكه‌ڤن. زمانێ مه‌ژى دیسان ئاسته‌نگه‌ك دى یه‌ و پارچه‌یه‌. ئه‌رێ مه‌ زمانه‌ك ره‌وشه‌نبیرى هه‌یه‌ تێرا تێگه‌هشتنا مه‌ بكه‌ و مه‌ ب هه‌ڤ بگه‌هینه‌؟
ب: ئه‌ز ب باوه‌رم، تێكلى و په‌یوه‌ندى، ب تایبه‌ت هه‌بوونا هه‌ڤناسین و هه‌ڤ هه‌مبێزكرنێ و تشتێن وه‌ها تو زارڤه‌یان ناسناكن. ژ بۆنى باكووریان سۆرانى زارڤه‌یه‌كا جودایه‌، ژ بۆنى باشووریان زازاكى چه‌ندى كوردى به‌ ژى ژ هێلا تێگه‌هشتنێ دا زه‌حمه‌ته‌. لێ ب تێكلىیا؛ ئه‌ز دوێ باوه‌رێ ده‌ مه‌ كو زاراڤه‌ نه‌بن سنۆر ئان ئاسته‌نگ د پێشبه‌رى تێگه‌هشتن و هه‌ڤ همبێزكرن و هه‌ڤدو ناسینێ ده‌.

ب: ئه‌ز دوێ باوه‌رێ ده‌ مه‌ و لێ ئیرۆ سنۆرێن كو دژمنى دناڤبه‌را كوردستانێ ده‌ كێشانه‌، بۆ نموونه‌ سنۆرێ مه‌، سنۆرێن كۆ دناڤبه‌را زمان، زمانێ كوردى ده‌. زاراڤه‌یێن جودا ژ بۆى من

           یانى (ئالفابێ) ئێك ژ پروبلمێ سه‌ره‌كى یه‌؟ لێ دیسان ژى پرۆبلمێ دن ژى هه‌نه‌، سنۆرێ وه‌سا چێكرنه‌ كو مرۆڤێ كورد ژ هه‌ڤ دانه‌ پاش. هوون ژ بۆ ڤێ چ دبێژن؟ نێرینێ وه‌ چىیه‌؟
ب: خه‌ونا چ سه‌رۆكى ژ خه‌ونا شاعره‌كى نه‌مه‌زن تره‌، خه‌ونا چ بزاڤێ ژ خه‌ونا هونه‌رمه‌نده‌كى نه‌گه‌وره‌تره‌. ئه‌ز ڤێ دبێژم: ل چار په‌رچێن كوردستانێ بهوسته‌ك ئاخا ئارژه‌ن ئارى نینه‌، لێ هه‌ر چار په‌رچێن كوردستانێ وه‌كا ملكێ دلێ منن. ئه‌ز ددلێ خوه‌دا دكارم وان باژۆم، دكارم وان بچینم و هه‌لینم. ئاسته‌نگى هه‌یه‌، راسته‌، كوردستانه‌كا ئازاد و سه‌ربخوه‌ چه‌ندى خه‌ونا هه‌موو هه‌لبه‌ستكار  و هونه‌رمه‌ند و نڤیسكاران به‌، دڤێ كو خه‌ونا هه‌ر كه‌سىبه‌  و هه‌ر سیاسه‌تمه‌داران ژى به‌، ئه‌ز دوێ خه‌ونێ ده‌ مه‌. لێ ناكوكیێن ناڤخۆیى ژ بۆى ده‌ستهه‌لاتداریا. ئان ژى بده‌ست خستنا كوردستانه‌كا دبن ده‌ستهه‌لاتداریا وان دا. وه‌كا هه‌لبه‌ستكاره‌كى ئه‌ز دێ هه‌ر و هه‌ر ل دژ بم.

          گه‌ره‌ك ل ڤر، ئه‌م نه‌ڕێنه‌كێ ل رۆلێ ره‌وشه‌نبیرێ كورد بكن. كانێ ره‌وشه‌نبیرێ كورد كارى یه‌ ل هه‌مبه‌ر ڤێ ده‌ستهه‌لاتێ، چ ده‌ستهه‌لاتا جڤاكێ به‌، چ ئولى به‌، ئان ژى سیاسى به‌ ده‌ركه‌ڤه‌؟ ده‌ركه‌ڤه‌ و بێژه‌ هوون شاشن و شاشتىیێ دكن؟ دكارن رێ بدن خه‌لكى كو گه‌ل بخوه‌ ببه‌ ده‌ستهه‌لات؟
ب: سه‌یدایێ هزرڤان: د سوحبه‌ته‌كێ دا من گۆت: ئه‌ز جوهى یێ كوردانم، د ده‌سپێكا دامه‌زاندنا وه‌لاته‌كى دا، یان ده‌ستهه‌لاتداریه‌كىدا مومكنه‌ ئه‌ز چاڤێ خوه‌ هنده‌ك تشتان بگرم، ژ بۆ چ؟ ژبۆنى كو دامه‌زراندن ئان رونشتنا ده‌ستگه‌هێن ده‌وله‌ته‌كێ د پێڤاژۆیه‌كىده‌ دڤێ هه‌ڕه‌. لێ ئیرۆ وه‌لاته‌ك بوویه‌ یه‌ك ئال، ب یه‌ك درشم، ب یه‌ك په‌رگال. بخوازن ژى راستیه‌ك هه‌یه‌، باشوور به‌رى سیه و پێنج سالان ژى هه‌ر قویبله‌گه‌ها كوردێن باكوور بوو. گاڤا كو دگۆتن به‌رزانى! سه‌رۆكێ هه‌موو كوردان دهاته‌ بیرێ، دهاته‌ بیرا مرۆڤى. دڤێ ره‌وشه‌نبیرێ مه‌ ب كاربن وێ رۆحێ دیسان ڤه‌گه‌رینن خه‌تێ كورد، ئه‌و گیانێ وانێ هندابووى،ئه‌ڤ ژى نه‌ ب رێكا سیاسه‌تێ دكارن، ئه‌ڤ ب رێكا كه‌سێن ره‌وشه‌نبیر و نڤیسكار تێ كرن.

          گۆتنه‌ك هه‌یه‌؛ جارنا تێ گۆتن ژ كوردان و ژ بلى كوردان، نه‌مازه‌ ئه‌و كوردێن كو ره‌وشه‌نبیرى یا گه‌له‌كى دى دهلگرن. دبێژن: مه‌ كوردان تو ره‌وشه‌نبیر نینن، مه‌ كوردان ره‌وشه‌نبیرى نینه‌! تو ژ بۆ ڤێ دیتنێ چ د بێژى؟ یان دیتنا ته‌ چىیه‌؟
ب: ئه‌گه‌ر مه‌به‌ستا وان ب ڤێ گۆتنێ مخالفه‌تكرنا ره‌وشه‌نبیران بێ، ئۆ هوون ژ من ب پرسن، ئیرۆ ره‌وشه‌نبیر رۆلا خوه‌یا مخالفه‌تكرنێ (به‌رهه‌نگار بوونێ) هێ نكارن ب لیزن. لێ گرنگه‌ بزانین د كیژان چارچۆڤه‌ى دا دكارن ب له‌یزن، ئه‌و كه‌سێن كو سه‌رۆكن، ئانى بزاڤن، یان ژى كوردستانه‌كا ئازاد و سه‌ربخوه‌ نه‌خوازن، ره‌وشه‌نبیر دڤێ دا دكارن مخالفه‌ت بكن، مخالفه‌تكرنا ئیرۆ ئه‌گه‌ر ژێ مه‌به‌ست ده‌ستهه‌لاتدارى بێ، ل هه‌موو وه‌لاتان ل ده‌ما دامه‌زراندنا ده‌وله‌ته‌كێدا مخالفه‌ت ده‌نگێ وێ كێمه‌. ئه‌ز باهوه‌ر دكم دپێشه‌رۆژێ ده‌ وێ ره‌وشه‌نبیرێ كورد، ب قاسى كو ده‌نگێ خوه‌ ل هه‌مبه‌رێ دژمنێ خوه‌ هلدایه‌، حه‌بس دیتنه‌، زیندان دیتنه‌ ولێدان خوارنه‌. مومكنه‌ كو ئه‌ز وه‌كا شاعره‌كى ژ هێلا زێده‌تریا خوه‌ دا ژى گه‌له‌ك دفكرم. ئه‌گه‌ر مه‌ به‌ست ژێ ئه‌و به‌، هێ ئه‌و ده‌م نه‌هاتى یه‌.

ب: ئه‌ڤ مژار، ئه‌ز باوه‌رم یه‌كه‌م جار د سالا (1992) دا دناڤا كوردێن سۆریێ ده‌ ده‌ستپێكر. یانى ئه‌ز خوه‌ وه‌كه‌ ره‌وشه‌نبیره‌كى نابینم، ژ به‌رچى؟ ژ به‌ر كو كارێ من هه‌لبه‌ستكارىیه‌، لێ راستیه‌ك ژى هه‌یه‌ گاڤا كو نڤیسكاره‌ك، یان هه‌لبه‌ستكاره‌ك پشتى نڤێساندنا هه‌لبه‌ستێن خوه‌، هه‌لبه‌ستێن خوه‌ خوه‌ندن و ده‌ركه‌ت پێش به‌رى جڤاكێ ئاخڤى، ل سه‌ر پرسێن كو ئاراسته‌ى وى بوون به‌رسڤان بدێ. بڤێ و نه‌ڤێ وێ گاڤێ لنگێ خوه‌ تاڤێ قادا ره‌وشه‌نبیرى یێ، رۆلێ ره‌وشه‌نبیرى دلیزه‌. ئه‌ڤ رۆل دكاره‌ رۆله‌كه‌ پڕ ژێهاتى و جوان به‌، ئه‌ڤ رۆل دكاره‌ پڕ لاوازه‌ به‌. راسته‌ ئیرۆ (ئێدوارد سه‌عید)ێ كوردان نینن، لێ ره‌وشه‌نبیرێن كوردان هه‌نه‌. تو نه‌ بوونا ره‌وشه‌نبیرێ كوردان ب تایبه‌ت ل باكوور، ئه‌ز دكارم وه‌كه‌ ته‌سبیته‌كێ بێژم ب قاسى كو سۆچى ده‌ستهه‌لاتداریا داگیركه‌رایه‌. هه‌و چه‌ند سیاسه‌تا ل سه‌ر ناڤێ كورد و كوردستان ژى تێ ئاژۆتن تاوانباره‌. ژ به‌رچى؟ ژبه‌ر كو ره‌وشه‌نبیران هه‌ر ئه‌رك تێ خوه‌ستن: مه‌ ب پارێزن، مه‌ بكن تاجا سه‌رێ خوه‌.

           ب گوڕا نه‌رینێ هه‌وه‌، گه‌ره‌ك ئه‌ڤ ژ ره‌وشه‌نبیر نه‌هێخوه‌ستن و ره‌وشه‌نبیر ژى داخوازا پارستنێ نه‌كه‌؟ به‌لكو دژ ده‌ركه‌ڤه‌؟
ب: یانى داخواز یا پارستنێ وه‌ك هوون دزانن په‌یڤه‌كا كوردان یا باڤ و كالان هه‌یه‌ دبێژن: (ده‌ست به‌رشۆ، ده‌ست به‌گه‌لوره‌ت شۆ) ئه‌گه‌ر بزاڤه‌كا سیاسى ل سه‌ر ناڤێ داهاتویا وه‌لاته‌كى، ئان ژى ئا توڤه‌كى، هونه‌رمه‌نده‌كى ئان نڤیسكاره‌كى مینا به‌رده‌ڤكێ خوه‌ ب كار بینه‌، ئه‌و نه‌ ره‌وشه‌نبیرىیه‌ و راست و راست خوه‌ فرۆتنه‌. وه‌ك هوون ژى باش دزانن ل وه‌لاتێن پێشكه‌تى و ل وه‌لاتێن پاشكه‌تى ژى به‌رده‌ڤك و لایه‌نگر و ره‌وشه‌نبیرێن ده‌ستهه‌لاتێ هه‌نه‌. ئه‌گه‌ر ژ من بخوه‌ ب پرسن، ئه‌ز وانا مینا قڕدكێن دیوانخانا سولتانێن به‌رێ دبینم.

           ل داویێ ئه‌م گه‌له‌ك سۆپاسیا وه‌ دكه‌ن و بخێرهاتنا وه‌ دكه‌ن ژ بۆى كوردستانا ئازاد، باوه‌ر دكم مه‌ هوون ماندى كرن.
ب: سۆپاس ژ بۆ هه‌وه‌ ژى و كۆڤارا وه‌ سه‌ركه‌تىبه‌.
